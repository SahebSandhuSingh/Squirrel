# Runtime dependencies for the Step 1 pose-recognition backend.
#
# Verified against the versions in the comments. MediaPipe is the one to watch:
# >=1.0 exposes ONLY the Tasks API (mediapipe.tasks.python.vision.PoseLandmarker),
# which is what pose_backend/pose_estimator.py uses. On 0.10.x the Tasks API
# exists too, but that line also still carries the old mp.solutions.pose API —
# if you pin 0.10.x, nothing here needs to change.
#
# SYSTEM PACKAGES (not pip-installable): MediaPipe >= 1.0 loads a native library
# that needs EGL/GLES present, even for CPU-only inference:
#     apt-get install -y libegl1 libgles2
# Without them, creating a PoseLandmarker fails with
# "OSError: libEGL.so.1: cannot open shared object file".

mediapipe>=1.0.0,<2          # verified: 1.0.1  (BlazePose, 33 landmarks)
numpy>=1.26,<2               # verified: 1.26.4
